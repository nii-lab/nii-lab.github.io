import RPi.GPIO as GPIO
import time

GPIO.setmode(GPIO.BCM)
GPIO.setup(5, GPIO.OUT)

try:
    while(True):
        GPIO.output(5, True)
        time.sleep(1)
        GPIO.output(5, False)
        time.sleep(1)
finally:
    print('cleaning up')
    GPIO.cleanup()
